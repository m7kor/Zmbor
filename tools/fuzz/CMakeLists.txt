# libFuzzer harnesses (docs/architecture/fuzzing.md). Configured only with PS2X_BUILD_FUZZERS=ON.
if(MSVC)
    set(PS2X_FUZZER_FLAG /fsanitize=fuzzer)
else()
    set(PS2X_FUZZER_FLAG -fsanitize=fuzzer)
endif()

function(ps2x_add_fuzzer name source target_id)
    add_executable(${name} ${source})
    target_compile_definitions(${name} PRIVATE PS2X_FUZZ_TARGET=${target_id})
    target_compile_options(${name} PRIVATE ${PS2X_FUZZER_FLAG})
    if(NOT MSVC)
        target_link_options(${name} PRIVATE ${PS2X_FUZZER_FLAG})
    else()
        target_link_options(${name} PRIVATE "/STACK:8388608")
    endif()
    target_compile_features(${name} PRIVATE cxx_std_20)
endfunction()

foreach(pair gif:1 vif0:2 vif1:3 dma:4)
    string(REPLACE ":" ";" pair "${pair}")
    list(GET pair 0 tname)
    list(GET pair 1 tid)
    ps2x_add_fuzzer(ps2x_fuzz_${tname} fuzz_runtime.cpp ${tid})
    target_include_directories(ps2x_fuzz_${tname} PRIVATE ${CMAKE_SOURCE_DIR}/ps2xRuntime/include)
    target_link_libraries(ps2x_fuzz_${tname} PRIVATE ps2_runtime ps2_iop)
endforeach()

foreach(pair elf:5 decoder:6)
    string(REPLACE ":" ";" pair "${pair}")
    list(GET pair 0 tname)
    list(GET pair 1 tid)
    ps2x_add_fuzzer(ps2x_fuzz_${tname} fuzz_recompiler.cpp ${tid})
    target_include_directories(ps2x_fuzz_${tname} PRIVATE ${CMAKE_SOURCE_DIR}/ps2xRecomp/include)
    target_link_libraries(ps2x_fuzz_${tname} PRIVATE ps2_recomp_lib)
endforeach()
